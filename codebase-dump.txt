from typing import Optional, Dict, Any

def atm_to_kpa(atm: Optional[float]) -> Optional[float]:
    if atm is None:
        return None
    return round(atm * 101.325, 2)

def fraction_to_percent(val: Optional[float]) -> Optional[float]:
    if val is None:
        return None
    return round(val * 100.0, 2) if val <= 1.0 else round(val, 2)

def generate_risk_assessment(
    o2_pct: Optional[float],
    airflow: Optional[float],
    fuel_category: str,
    cool_flame_k: Optional[float] = None
) -> tuple[str, str]:
    """Translates complex combustion telemetry into a clear risk badge and explanation."""
    if cool_flame_k is not None:
        return (
            "High",
            f"Invisible Cool-Flame Regime: Continues burning at {cool_flame_k} mm²/s after visible flame extinction, evading standard optical sensors."
        )
    if o2_pct and o2_pct >= 25.0:
        return (
            "Critical",
            f"Exploration Atmosphere ({o2_pct}% O₂): Elevated oxygen lowers ignition threshold and increases flame size/deflagration risk."
        )
    if airflow is not None and airflow <= 5.0 and "Solid" in fuel_category:
        return (
            "High",
            "Low-Flow Vapour Accumulation: Reduced ventilation (≤5 cm/s) traps unburned pyrolysis gases, creating sudden deflagration risk."
        )
    return (
        "Moderate",
        "Controlled Microgravity Burn: Flame reaches a bounded steady size governed by forced ventilation rather than buoyancy."
    )

def format_measurement_card(run: Dict[str, Any]) -> Dict[str, Any]:
    """Refines a raw DB row into a frontend-ready object with explicit units and nulls preserved."""
    measurements = []
    if run.get("oxygen_percent") is not None:
        measurements.append({"label": "Oxygen Level", "value": run["oxygen_percent"], "unit": "%"})
    if run.get("pressure_kpa") is not None:
        measurements.append({"label": "Chamber Pressure", "value": run["pressure_kpa"], "unit": "kPa"})
    if run.get("airflow_cm_s") is not None:
        measurements.append({"label": "Forced Airflow", "value": run["airflow_cm_s"], "unit": "cm/s"})
    if run.get("steady_flame_size_cm") is not None:
        measurements.append({"label": "Steady Flame Size", "value": run["steady_flame_size_cm"], "unit": "cm"})
    if run.get("max_flame_size_cm") is not None:
        measurements.append({"label": "Peak Flame Size", "value": run["max_flame_size_cm"], "unit": "cm"})
    if run.get("burn_duration_s") is not None:
        measurements.append({"label": "Burn Duration", "value": run["burn_duration_s"], "unit": "s"})
    if run.get("hot_flame_k_mm2_s") is not None:
        measurements.append({"label": "Hot Burning Rate (K)", "value": run["hot_flame_k_mm2_s"], "unit": "mm²/s"})
    if run.get("cool_flame_k_mm2_s") is not None:
        measurements.append({"label": "Cool Burning Rate (K)", "value": run["cool_flame_k_mm2_s"], "unit": "mm²/s"})
    if run.get("avg_heat_release_w") is not None:
        measurements.append({"label": "Avg Heat Release", "value": run["avg_heat_release_w"], "unit": "W"})
    if run.get("co2_increase_ppm") is not None:
        measurements.append({"label": "Cabin CO₂ Rise", "value": run["co2_increase_ppm"], "unit": "ppm"})
    if run.get("co_increase_ppm") is not None:
        measurements.append({"label": "Cabin CO Rise", "value": run["co_increase_ppm"], "unit": "ppm"})
    if run.get("limiting_oxygen_index") is not None:
        measurements.append({"label": "Limiting O₂ Index", "value": run["limiting_oxygen_index"], "unit": "mol frac"})

    return {
        **run,
        "display_measurements": measurements,
        "disclaimer": f"Verified NASA record ({run['source_reference']}). Unreported channels omitted rather than zeroed."
    }

import argparse
import pandas as pd
from supabase import create_client
from app.config import settings
from app.cleaner import atm_to_kpa, fraction_to_percent, generate_risk_assessment

supabase = create_client(settings.SUPABASE_URL, settings.SUPABASE_SERVICE_ROLE_KEY)

def ingest_psi_file(filepath: str, investigation_id: str, source_citation: str):
    """
    Cleans raw NASA PSI CSV/Excel exports (e.g., PSI-25 BASS-II or PSI-98 FLEX),
    standardizes units (atm -> kPa, mole fraction -> %), replaces NaNs with None,
    and upserts into Supabase experiment_runs.
    """
    df = pd.read_csv(filepath) if filepath.endswith(".csv") else pd.read_excel(filepath)
    df = df.where(pd.notnull(df), None)

    cleaned_rows = []
    for idx, row in df.iterrows():
        o2_raw = row.get("oxygen_percent") or row.get("O2") or row.get("o2_mole_fraction")
        o2_pct = fraction_to_percent(float(o2_raw)) if o2_raw is not None else None

        p_kpa = row.get("pressure_kpa")
        if p_kpa is None and row.get("pressure_atm") is not None:
            p_kpa = atm_to_kpa(float(row["pressure_atm"]))

        airflow = float(row["airflow_cm_s"]) if row.get("airflow_cm_s") is not None else None
        cool_k = float(row["cool_flame_k_mm2_s"]) if row.get("cool_flame_k_mm2_s") is not None else None
        fuel_cat = str(row.get("fuel_category") or "Thin Solid")

        risk, summary = generate_risk_assessment(o2_pct, airflow, fuel_cat, cool_k)

        cleaned_rows.append({
            "investigation_id": investigation_id.upper(),
            "run_code": str(row.get("run_code") or row.get("test_point") or f"RUN-{idx+1}"),
            "fuel_material": str(row.get("fuel_material") or row.get("fuel") or "Unknown NASA Sample"),
            "fuel_category": fuel_cat,
            "sample_geometry": str(row["sample_geometry"]) if row.get("sample_geometry") else None,
            "oxygen_percent": o2_pct,
            "pressure_kpa": float(p_kpa) if p_kpa is not None else None,
            "airflow_cm_s": airflow,
            "diluent_gas": str(row["diluent_gas"]) if row.get("diluent_gas") else None,
            "flow_direction": str(row["flow_direction"]) if row.get("flow_direction") else None,
            "steady_flame_size_cm": float(row["steady_flame_size_cm"]) if row.get("steady_flame_size_cm") is not None else None,
            "max_flame_size_cm": float(row["max_flame_size_cm"]) if row.get("max_flame_size_cm") is not None else None,
            "burn_duration_s": float(row["burn_duration_s"]) if row.get("burn_duration_s") is not None else None,
            "hot_flame_k_mm2_s": float(row["hot_flame_k_mm2_s"]) if row.get("hot_flame_k_mm2_s") is not None else None,
            "cool_flame_k_mm2_s": cool_k,
            "plain_english_outcome": str(row.get("plain_english_outcome") or summary),
            "risk_level": risk,
            "instrument_source": str(row.get("instrument_source") or "NASA PSI Telemetry"),
            "source_reference": source_citation,
        })

    if cleaned_rows:
        supabase.table("experiment_runs").upsert(
            cleaned_rows, on_conflict="investigation_id,run_code"
        ).execute()
    print(f"✅ Ingested and cleaned {len(cleaned_rows)} runs into {investigation_id.upper()}.")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Clean and upload NASA PSI CSV/Excel files to Supabase")
    parser.add_argument("filepath", help="Path to CSV or XLSX file")
    parser.add_argument("--investigation", required=True, help="Investigation ID (e.g., BASS-II, FLEX, SAFFIRE-IV)")
    parser.add_argument("--citation", default="NASA PSI Repository", help="Official NASA citation")
    args = parser.parse_args()
    ingest_psi_file(args.filepath, args.investigation, args.citation)

@app.get("/api/kpis")
def get_dashboard_kpis():
    """Powers kpi-cards.tsx with verified cross-mission summary metrics."""
    runs = supabase.table("experiment_runs").select("*").execute().data
    investigations = supabase.table("investigations").select("id, si_score").execute().data
    
    valid_o2 = [r["oxygen_percent"] for r in runs if r.get("oxygen_percent") is not None]
    valid_hrr = [r["avg_heat_release_w"] for r in runs if r.get("avg_heat_release_w") is not None]
    avg_si = round(sum(i["si_score"] for i in investigations) / len(investigations), 1) if investigations else 0

    return {
        "total_investigations": len(investigations),
        "verified_benchmark_runs": len(runs),
        "average_si_score": avg_si,
        "lowest_tested_o2_percent": min(valid_o2) if valid_o2 else None,
        "peak_heat_release_w": max(valid_hrr) if valid_hrr else None,
        "microgravity_gravity_level": "10⁻⁶ g (ISS / Cygnus)"
    }

@app.get("/api/investigations")
def list_investigations(
    destination: Optional[str] = Query(None, description="Filter by ISS, Moon, Mars"),
    search: Optional[str] = Query(None, description="Search by name, program, fuel, or tag")
):
    """Powers experiment-ranking.tsx leaderboard & investigation detail panel."""
    rows = supabase.table("investigations").select("*").order("si_score", desc=True).execute().data
    if destination and destination.lower() != "all":
        rows = [r for r in rows if destination in (r.get("destinations") or [])]
    if search:
        q = search.strip().lower()
        rows = [
            r for r in rows
            if q in r["full_name"].lower()
            or q in r["program"].lower()
            or q in r["fuel_summary"].lower()
            or any(q in t.lower() for t in (r.get("tags") or []))
        ]
    return rows

@app.get("/api/runs", response_model=List[RefinedExperimentRun])
def get_refined_runs(
    investigation_id: Optional[str] = Query(None, description="Filter by BASS, BASS-II, FLEX, CFI, SAFFIRE-I..V, SAME"),
    fuel_category: Optional[str] = Query(None, description="Filter by Thin Solid, Thick Solid, Liquid Droplet, Aerosol/Overheated Material"),
    min_o2: Optional[float] = Query(None),
    max_o2: Optional[float] = Query(None),
):
    """Returns cleaned NASA runs with explicit units and plain-English outcomes."""
    query = supabase.table("experiment_runs").select("*")
    if investigation_id:
        query = query.eq("investigation_id", investigation_id.upper())
    if fuel_category:
        query = query.eq("fuel_category", fuel_category)
    if min_o2 is not None:
        query = query.gte("oxygen_percent", min_o2)
    if max_o2 is not None:
        query = query.lte("oxygen_percent", max_o2)

    raw_runs = query.execute().data
    return [format_measurement_card(r) for r in raw_runs]

@app.get("/api/charts/{chart_group}")
def get_chart_series(chart_group: str):
    """
    Powers spread-chart.tsx & droplet decay graphs.
    Available groups: 'spread_vs_oxygen', 'spread_vs_gravity', 'airflow_vs_flame_size', 'droplet_d2_decay'
    """
    res = (
        supabase.table("experiment_timeseries")
        .select("*, experiment_runs(run_code, investigation_id, fuel_material, source_reference)")
        .eq("chart_group", chart_group)
        .order("x_value")
        .execute()
    )
    if not res.data:
        raise HTTPException(status_code=404, detail=f"No chart series found for '{chart_group}'")

    # Pivot multi-material series (pmma, cotton, nomex) cleanly for Recharts LineChart
    if chart_group in ["spread_vs_oxygen", "spread_vs_gravity"]:
        x_field = "o2" if chart_group == "spread_vs_oxygen" else "g"
        pivoted = {}
        for pt in res.data:
            x_val = str(pt["x_value"]) if x_field == "g" else pt["x_value"]
            if x_val not in pivoted:
                pivoted[x_val] = {x_field: x_val}
            if pt.get("series_key"):
                pivoted[x_val][pt["series_key"]] = pt["y_value"]
        return list(pivoted.values())

    return res.data

@app.get("/api/simulate", response_model=SimulationResponse)
def simulate_microgravity_fire(
    gravity_g: float = Query(0.0, ge=0.0, le=1.0, description="0 = ISS, 0.16 = Moon, 0.38 = Mars, 1.0 = Earth"),
    oxygen_percent: float = Query(21.0, ge=10.0, le=40.0),
    airflow_cm_s: float = Query(5.0, ge=0.0, le=50.0),
    fuel_type: str = Query("SIBAL", description="SIBAL, PMMA, n-Decane, Heptane")
):
    """
    Powers flame-simulator.tsx, returning both frontend visual metrics (temp, spread, soot, risk)
    and NASA empirical safety guidance.
    """
    peak = 0.0 if gravity_g == 0 else (gravity_g * math.exp(1 - gravity_g / 0.2)) / 0.2
    o2_factor = (oxygen_percent / 21.0) ** 2
    flow_factor = 0.35 if (gravity_g < 0.05 and airflow_cm_s == 0) else (1.0 + airflow_cm_s / 40.0)
    spread = round((1.2 + 2.2 * peak + 0.6 * gravity_g) * o2_factor * flow_factor, 2)
    temp = round(1400 + gravity_g * 500 + (oxygen_percent - 21.0) * 25 + airflow_cm_s * 6)
    soot = round(min(100, max(5, 25 + gravity_g * 45 + (oxygen_percent - 21.0) * 2.5 + airflow_cm_s * 0.4)))
    risk_score = min(100, max(0, round(spread * 11 + (oxygen_percent - 21.0) * 2)))
    risk_label = "High" if risk_score > 66 else ("Elevated" if risk_score > 33 else "Low")

    if fuel_type.lower() in ["heptane", "n-decane"]:
        loi = 15.0
        if oxygen_percent < loi:
            regime = "Extinguished (Below LOI)"
            advice = "Below the 15% O₂ Limiting Oxygen Index in N₂; neither hot nor steady cool flame sustains."
        elif oxygen_percent < 18.0:
            regime = "Cool-Flame Transition Risk"
            advice = "Hot flame may radiatively extinguish early, leaving an invisible cool flame burning (FLEX/CFI finding)."
        else:
            regime = "Quiescent Spherical Burn"
            advice = "Sustains two-stage burning (hot flame followed by invisible cool flame)."
        return SimulationResponse(
            fuel=fuel_type, gravity_g=gravity_g, oxygen_percent=oxygen_percent, airflow_cm_s=airflow_cm_s,
            regime=regime, flame_temp_k=temp, spread_rate_mm_s=spread, soot_index=soot,
            risk_score=risk_score, risk_label=risk_label, predicted_steady_flame_cm=None,
            safety_advice=advice, model_basis="NASA FLEX/CFI droplet empirical bounds"
        )

    if gravity_g < 0.05 and airflow_cm_s <= 5.0 and oxygen_percent >= 21.0:
        regime = "Deflagration / Vapour Accumulation Hazard"
        estimated_size = round(3.0 + (oxygen_percent - 21.0) * 0.8, 1)
        advice = "NASA Saffire-IV/V warning: Dropping flow to ≤5 cm/s in microgravity traps hot pyrolysis vapours and can trigger rapid deflagration."
    elif oxygen_percent < 16.0:
        regime = "Quenched / Non-Sustaining"
        estimated_size = 0.0
        advice = "Oxygen below typical microgravity flame-spread threshold."
    else:
        regime = "Bounded Microgravity Flame" if gravity_g < 0.1 else "Buoyancy-Assisted Flame"
        estimated_size = round(4.0 * max(airflow_cm_s, 5.0) / 20.0 * (oxygen_percent / 21.0) ** 1.8, 2)
        advice = "In microgravity, flames reach a bounded steady length controlled by forced ventilation rather than buoyancy."

    return SimulationResponse(
        fuel=fuel_type, gravity_g=gravity_g, oxygen_percent=oxygen_percent, airflow_cm_s=airflow_cm_s,
        regime=regime, flame_temp_k=temp, spread_rate_mm_s=spread, soot_index=soot,
        risk_score=risk_score, risk_label=risk_label, predicted_steady_flame_cm=estimated_size,
        safety_advice=advice, model_basis="Empirical interpolation from NASA Saffire I–V & BASS-II"
    )

-- 1. Master Investigations Table (Powering Experiment Leaderboard & Investigation Cards)
CREATE TABLE investigations (
    id TEXT PRIMARY KEY,                      -- e.g., 'BASS', 'BASS-II', 'FLEX', 'CFI', 'SAFFIRE-I', 'SAFFIRE-II', 'SAFFIRE-III', 'SAFFIRE-IV', 'SAFFIRE-V', 'SAME'
    family TEXT NOT NULL,                     -- e.g., 'BASS Family', 'Saffire Series', 'Droplet Combustion', 'Aerosol'
    full_name TEXT NOT NULL,
    program TEXT NOT NULL,                    -- e.g., 'Spacecraft Fire Safety', 'Burning and Suppression of Solids'
    platform TEXT NOT NULL,                   -- e.g., 'ISS · MSG', 'ISS · CIR', 'Cygnus OA-6 (uncrewed)'
    years TEXT NOT NULL,                      -- e.g., '2016', '2012–2021', '2009–2014'
    fuel_summary TEXT NOT NULL,               -- e.g., 'SIBAL cotton-fiberglass, PMMA, Nomex'
    si_score INTEGER NOT NULL CHECK (si_score BETWEEN 0 AND 100),
    destinations TEXT[] NOT NULL,             -- e.g., ARRAY['ISS', 'Moon', 'Mars']
    tags TEXT[] NOT NULL,                     -- e.g., ARRAY['Large-scale', 'Flame spread', 'Cabin']
    primary_focus TEXT NOT NULL,
    summary_plain_english TEXT NOT NULL,      -- Simplified explanation for normal users
    key_safety_takeaway TEXT NOT NULL,
    psi_url TEXT,
    ntrs_citation TEXT
);

-- 2. Individual Experimental Runs / Test Points (Verified from NASA PSI & NTRS Briefs)
CREATE TABLE experiment_runs (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    investigation_id TEXT REFERENCES investigations(id) ON DELETE CASCADE,
    run_code TEXT NOT NULL,                   -- e.g., 'R377', 'I-1', 'II-1', 'III-1', 'IV-1', 'V-2'
    fuel_material TEXT NOT NULL,              -- e.g., 'SIBAL Cotton-Fiberglass', 'PMMA (Acrylic)', 'n-Decane', 'Silicone', 'Teflon'
    fuel_category TEXT NOT NULL,              -- 'Thin Solid', 'Thick Solid', 'Liquid Droplet', 'Gas', 'Aerosol/Overheated Material'
    sample_geometry TEXT,                     -- e.g., '94cm x 40.6cm sheet (0.37mm)', 'Droplet D0=4.03mm', 'Ribbed 10mm slab'
    
    -- Standardized Environmental Conditions (NULL = Not Reported; never zeroed)
    oxygen_percent NUMERIC(5,2),              -- mol% or vol%
    pressure_kpa NUMERIC(6,2),                -- Standardized to kPa (1 atm = 101.325 kPa)
    airflow_cm_s NUMERIC(6,2),                -- cm/s (0 = quiescent)
    diluent_gas TEXT,                         -- e.g., 'N2', 'CO2', 'He'
    flow_direction TEXT,                      -- 'Concurrent', 'Opposed', 'Quiescent/Radial'
    
    -- Refined Outcomes & Measurements
    flame_spread_rate_mm_s NUMERIC(8,3),
    steady_flame_size_cm NUMERIC(6,2),
    max_flame_size_cm NUMERIC(6,2),
    burn_duration_s NUMERIC(8,2),
    hot_flame_k_mm2_s NUMERIC(6,3),           -- Droplet burning-rate constant (Hot)
    cool_flame_k_mm2_s NUMERIC(6,3),          -- Droplet burning-rate constant (Cool)
    avg_heat_release_w NUMERIC(8,2),
    co2_increase_ppm NUMERIC(8,2),
    co_increase_ppm NUMERIC(8,2),
    limiting_oxygen_index NUMERIC(5,3),       -- LOI fraction (e.g., 0.150)
    
    -- Human-Readable Interpretation & Provenance
    plain_english_outcome TEXT NOT NULL,
    risk_level TEXT CHECK (risk_level IN ('Low', 'Moderate', 'High', 'Critical')),
    instrument_source TEXT,
    source_reference TEXT NOT NULL,
    
    UNIQUE(investigation_id, run_code)
);

-- 3. Time-Series & Graph Points (For Spread Charts, Cool-Flame Transitions, Particle Distributions)
CREATE TABLE experiment_timeseries (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    run_id UUID REFERENCES experiment_runs(id) ON DELETE CASCADE,
    chart_group TEXT NOT NULL,                -- 'airflow_vs_flame_size', 'droplet_d2_decay', 'spread_vs_oxygen', 'spread_vs_gravity'
    x_label TEXT NOT NULL,
    x_value NUMERIC(10,3) NOT NULL,
    y_label TEXT NOT NULL,
    y_value NUMERIC(10,3) NOT NULL,
    series_key TEXT,                          -- e.g., 'pmma', 'cotton', 'nomex', 'steady', 'peak'
    regime_tag TEXT,
    unit_note TEXT
);

-- 4. Habitat Fire Risk Matrix (Powering habitat-radar.tsx)
CREATE TABLE habitat_risk_profiles (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    factor TEXT UNIQUE NOT NULL,
    iss INTEGER NOT NULL CHECK (iss BETWEEN 0 AND 100),
    gateway INTEGER NOT NULL CHECK (gateway BETWEEN 0 AND 100),
    lunar INTEGER NOT NULL CHECK (lunar BETWEEN 0 AND 100),
    mars INTEGER NOT NULL CHECK (mars BETWEEN 0 AND 100),
    scientific_basis TEXT NOT NULL
);

-- 5. Curated Q&A Knowledge Base (Powering si-assistant.tsx)
CREATE TABLE si_knowledge_qa (
    id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
    question TEXT UNIQUE NOT NULL,
    answer TEXT NOT NULL,
    source_citations TEXT NOT NULL
);

# Flame in Freefall — FastAPI + Supabase Backend

Complete backend and PostgreSQL (Supabase) database system verified against NASA PSI, NTRS, and the NASA research briefs (`BASS/BASS-II`, `FLEX/FLEX-2`, `CFI/CFI-G`, `Saffire I–V`, `SAME/SAME-R`).

## Quick Setup

1. **Create Supabase Tables**:
   - Open your Supabase project -> **SQL Editor**.
   - Paste and run `database/schema.sql`.

2. **Configure Environment**:
   - Copy `.env.example` to `.env` and add your `SUPABASE_URL` and `SUPABASE_SERVICE_ROLE_KEY`.

3. **Install Dependencies & Seed Verified Data**:
   ```bash
   python -m venv venv
   source venv/bin/activate  # Windows: venv\Scripts\activate
   pip install -r requirements.txt
   python -m app.seed_nasa_data
   ```

4. **Ingest Additional Raw PSI CSV/Excel Files (Optional)**:
   ```bash
   python -m app.ingest_psi_csv path/to/psi_data.csv --investigation BASS-II
   ```

5. **Run FastAPI Server**:
   ```bash
   uvicorn app.main:app --reload --port 8000
   ```
   Interactive API docs: `http://localhost:8000/docs`

function generateAIResponse(query) {
  const q = query.toLowerCase();
  if (q.includes("spherical") || q.includes("shape") || q.includes("round") || q.includes("ball")) {
    return "On Earth (1g), hot exhaust gases are less dense and rise vigorously via buoyancy (Grashof number Gr ~ 10⁵–10⁶), pulling fresh oxygen into the base and stretching the flame into an elongated flickering teardrop. In microgravity (10⁻⁶g), buoyant natural convection vanishes (Gr → 0). Oxygen and fuel mix solely through isotropic molecular diffusion (D ~ 0.2 cm²/s), forming a symmetric blue spherical flamelet that burns cooler and more quietly.";
  }
  if (q.includes("airflow") || q.includes("ventilation") || q.includes("cabin") || q.includes("fan")) {
    return "Spacecraft ECLSS ventilation (typically 15–25 cm/s) acts as low-speed forced convection. As revealed in NASA's BASS and Saffire experiments, stagnant 0g flames often self-extinguish in their own CO₂ blanket; however, gentle cabin airflow sweeps away exhaust products and delivers fresh O₂—creating a critical peak flame-spread regime between 10 and 25 cm/s. Therefore, emergency spacecraft procedures immediately shutdown cabin fans upon smoke confirmation.";
  }
  if (q.includes("cool flame") || q.includes("flex")) {
    return "During NASA's Flame Extinguishment Experiment (FLEX) on the ISS, researchers discovered that alkane droplets (such as n-heptane) visibly extinguished due to radiative heat loss, yet continued to burn invisibly at ~550–750 K. These two-stage 'cool flames' are sustained by low-temperature alkylperoxy radical chemistry without visible soot luminescence, producing toxic CO and formaldehyde.";
  }
  if (q.includes("extinguish") || q.includes("suppress") || q.includes("water") || q.includes("halon")) {
    return "In spacecraft cabins, fire suppression follows a mandatory 3-step protocol: (1) Cut electrical power to the affected equipment rack to isolate the ignition source, (2) Shut down ECLSS cabin ventilation fans to starve the fire of convective oxygen, and (3) Discharge fine water mist (Orion/Gateway exploration standard) or CO₂/N₂ inert gas (ISS standard) to drop local oxygen concentration below the Limiting Oxygen Index (~14%).";
  }
  if (q.includes("soot") || q.includes("acme") || q.includes("smoke") || q.includes("same")) {
    return "NASA's ACME and SAME investigations confirmed that microgravity diffusion flames have much longer residence times, yielding 5× to 8× more soot volume and 2× to 4× larger smoke aggregates than terrestrial flames. As a result, deep-space habitats rely on dual-wavelength photoelectric smoke scatter detectors rather than terrestrial ionization detectors.";
  }
  if (q.includes("sofie") || q.includes("gateway") || q.includes("34%")) {
    return "NASA's SoFIE investigation examines solid material flammability in exploration atmospheres. The Artemis Lunar Gateway operates at 56.5 kPa with 34% O₂. At this elevated oxygen level, ignition delay times decrease by over 70%, and materials like acrylic and fabrics ignite far more easily than under normal Earth 21% O₂ conditions.";
  }
  return "Based on NASA PSI and NTRS microgravity combustion datasets (FLEX, ACME, BASS, Saffire, SAME, and SoFIE): eliminating buoyant convection makes flames diffusion-controlled, spherical in stagnant air, and exceptionally sensitive to low-velocity cabin ventilation (10–30 cm/s) and exploration oxygen levels (up to 34% O₂).";

function appendChatMessage(sender, text) {
  const box = document.getElementById("chatMessages");
  if (!box) return;

  const div = document.createElement("div");
  div.className = `msg ${sender}`;

  if (sender === "user") {
    div.innerHTML = `<strong>Astronaut / Researcher:</strong> ${text}`;
    box.appendChild(div);
    box.scrollTop = box.scrollHeight;
  } else {
    // Typewriter effect for AI response
    div.innerHTML = `<strong>PSI Combustion Assistant:</strong> <span class="typing-text"></span><span class="cursor-blink">▌</span>`;
    box.appendChild(div);
    box.scrollTop = box.scrollHeight;

    const textSpan = div.querySelector(".typing-text");
    const cursor = div.querySelector(".cursor-blink");
    let charIdx = 0;

    function typeChar() {
      if (charIdx < text.length) {
        textSpan.textContent += text.charAt(charIdx);
        charIdx++;
        box.scrollTop = box.scrollHeight;
        setTimeout(typeChar, 14);
      } else {
        cursor.remove();
        // Add Copy Button
        const copyBtn = document.createElement("button");
        copyBtn.className = "copy-msg-btn";
        copyBtn.innerHTML = `📋 Copy`;
        copyBtn.onclick = () => {
          navigator.clipboard.writeText(text);
          copyBtn.innerHTML = `✓ Copied`;
          setTimeout(() => (copyBtn.innerHTML = `📋 Copy`), 2000);
        };
        div.appendChild(copyBtn);
      }
    }
    typeChar();
  }
}

// Interactive Dimensionless Scaling Calculator
window.updateDimensionlessCalculator = function () {
  const uSlider = document.getElementById("calcVelocitySlider");
  const lSlider = document.getElementById("calcLengthSlider");
  const gSelect = document.getElementById("calcGravitySelect");

  if (!uSlider || !lSlider || !gSelect) return;

  const U = parseFloat(uSlider.value) / 100; // cm/s to m/s
  const L = parseFloat(lSlider.value) / 100; // cm to m
  const gRatio = parseFloat(gSelect.value); // 0, 0.16, 0.38, 1.0
  const g = gRatio * 9.81;

  document.getElementById("calcVelocityVal").textContent = `${uSlider.value} cm/s`;
  document.getElementById("calcLengthVal").textContent = `${lSlider.value} cm`;

  const nu = 1.6e-5; // kinematic viscosity of air (m²/s)
  const beta = 1 / 300; // volumetric expansion coeff (1/K)
  const deltaT = 1100; // K

  // Grashof Number Gr = g * beta * deltaT * L³ / nu²
  const Gr = (g * beta * deltaT * Math.pow(L, 3)) / Math.pow(nu, 2);

  // Reynolds Number Re = U * L / nu
  const Re = (U * L) / nu;

  // Froude Number Fr = U² / (g * L)
  const Fr = g > 0 ? Math.pow(U, 2) / (g * L) : 9999;

  // Mixed convection criterion: Richardson number Ri = Gr / Re²
  const Ri = Re > 0 ? Gr / Math.pow(Re, 2) : (Gr > 0 ? 9999 : 0);

  document.getElementById("calcGrVal").textContent = Gr < 1 ? Gr.toExponential(2) : Math.round(Gr).toLocaleString();
  document.getElementById("calcReVal").textContent = Re.toFixed(1);
  document.getElementById("calcFrVal").textContent = Fr > 999 ? "∞ (0g)" : Fr.toFixed(2);

  const regimeBadge = document.getElementById("calcRegimeBadge");
  if (gRatio === 0) {
    if (U === 0) {
      regimeBadge.textContent = "Pure Molecular Diffusion (Gr = 0, Re = 0)";
      regimeBadge.style.color = "#00f5ff";
    } else {
      regimeBadge.textContent = "Forced Convection Only (Gr = 0, Ri = 0)";
      regimeBadge.style.color = "#38bdf8";
    }
  } else if (Ri > 10) {
    regimeBadge.textContent = "Buoyant Natural Convection Dominant (Gr/Re² >> 1)";
    regimeBadge.style.color = "#ff7b00";
  } else if (Ri >= 0.1 && Ri <= 10) {
    regimeBadge.textContent = "Mixed Convection Regime (Buoyancy ~ Forced Flow)";
    regimeBadge.style.color = "#fbbf24";
  } else {
    regimeBadge.textContent = "Forced Flow Dominant (Ventilation Overpowers Buoyancy)";
    regimeBadge.style.color = "#10b981";
  }
};

window.sortSITable = function (col) {
  if (siSortColumn === col) {
    siSortAsc = !siSortAsc;
  } else {
    siSortColumn = col;
    siSortAsc = true;
  }
  renderSIRanking();
};

window.renderSIRanking = function () {
  const filterEl = document.getElementById("siFilterSelect");
  const searchEl = document.getElementById("siSearchInput");
  const tbody = document.getElementById("siTableBody");
  if (!tbody) return;

  const filter = filterEl ? filterEl.value : "all";
  const search = searchEl ? searchEl.value.toLowerCase().trim() : "";

  let rows = filter === "all" ? [...SI_RANKING_DATA] : SI_RANKING_DATA.filter((item) => item.category === filter);

  if (search) {
    rows = rows.filter(
      (r) =>
        r.name.toLowerCase().includes(search) ||
        r.category.toLowerCase().includes(search) ||
        r.rating.toLowerCase().includes(search)
    );
  }

  // Sort rows
  rows.sort((a, b) => {
    let valA = a[siSortColumn];
    let valB = b[siSortColumn];
    if (typeof valA === "string") valA = valA.toLowerCase();
    if (typeof valB === "string") valB = valB.toLowerCase();
    if (valA < valB) return siSortAsc ? -1 : 1;
    if (valA > valB) return siSortAsc ? 1 : -1;
    return 0;
  });

  tbody.innerHTML = rows
    .map(
      (r) => `
    <tr>
      <td><strong>#${r.rank}</strong></td>
      <td><strong>${r.name}</strong></td>
      <td><span class="tag tag-${r.category.toLowerCase()}">${r.category}</span></td>
      <td>${r.loi}</td>
      <td>${r.spread}</td>
      <td><strong style="color:${r.score >= 80 ? "#34d399" : r.score >= 55 ? "#fbbf24" : "#f87171"}">${r.score} / 100</strong></td>
      <td>${r.rating}</td>
    </tr>
  `
    )
    .join("");

  const barCanvas = document.getElementById("siBarChart");
  if (!barCanvas) return;
  if (siBarInstance) siBarInstance.destroy();

  siBarInstance = new Chart(barCanvas.getContext("2d"), {
    type: "bar",
    data: {
      labels: rows.map((r) => r.name),
      datasets: [
        {
          label: "Safety Index (SI Score)",
          data: rows.map((r) => r.score),
          backgroundColor: rows.map((r) =>
            r.score >= 80 ? "rgba(16, 185, 129, 0.85)" : r.score >= 55 ? "rgba(245, 158, 11, 0.85)" : "rgba(239, 68, 68, 0.85)"
          ),
          borderColor: rows.map((r) => (r.score >= 80 ? "#10b981" : r.score >= 55 ? "#f59e0b" : "#ef4444")),
          borderWidth: 1.5,
          borderRadius: 6
        }
      ]
    },
    options: {
      responsive: true,
      plugins: { legend: { display: false } },
      scales: {
        x: { ticks: { color: "#e2e8f0", maxRotation: 25 }, grid: { display: false } },
        y: { min: 0, max: 100, ticks: { color: "#e2e8f0" }, grid: { color: "rgba(255, 255, 255, 0.08)" } }
      }
    }
  });
};

            <div class="tech-corner tl">+</div>
            <h4>1. Grashof & Froude Number Scaling</h4>
            <div class="formula-box">
              Gr = (g · β · ΔT · L³) / ν² &nbsp;&nbsp;|&nbsp;&nbsp; Fr = U² / (g · L)
            </div>
            <p>
              In terrestrial flames, the <strong>Grashof number (Gr ~ 10⁴–10⁶)</strong> dominates, generating buoyant velocities U<sub>b</sub> ≈ √(g·L·ΔT/T<sub>∞</sub>) ~ 0.5–1.5 m/s. On the ISS (g ≈ 10⁻⁶g₀), Gr → 0, collapsing natural convection. When cabin ventilation U<sub>air</sub> is introduced, the mixed-convection ratio Gr / Re² determines whether forced airflow or diffusion governs flame geometry.
            </p>
          </div>

          <div class="card theory-card">
            <div class="tech-corner tl">+</div>
            <h4>2. Damköhler Number & Radiative Extinction</h4>
            <div class="formula-box">
              Da = τ<sub>flow</sub> / τ<sub>chem</sub> = (L / U<sub>diff</sub>) / τ<sub>chem</sub>
            </div>
            <p>
              Because molecular diffusion is slow, characteristic residence times τ<sub>flow</sub> increase dramatically in microgravity. While high residence time initially boosts soot nucleation, expanded spherical flame surface area increases volumetric radiative heat loss (Q̇<sub>rad</sub> ∝ T⁴·V<sub>flame</sub>). At large droplet diameters, radiative loss drops temperature below the hot-flame threshold, triggering extinction or transition into a low-temperature <strong>Cool Flame</strong>.
            </p>
          </div>

          <div class="card theory-card">
            <div class="tech-corner tl">+</div>
            <h4>3. Lewis Number & Diffusive-Thermal Instability</h4>
            <div class="formula-box">
              Le = α / D<sub>mass</sub> = k / (ρ · c<sub>p</sub> · D<sub>mass</sub>)
            </div>
            <p>
              In lean hydrogen or methane mixtures (Le &lt; 1), mass diffusivity of the deficient reactant exceeds thermal diffusivity. In zero-g, this triggers cellular flame structures and isolated <strong>flame balls</strong> (studied in NASA SOFBALL) that remain stable far below terrestrial lean flammability limits.
            </p>
          </div>

# NASA Microgravity Combustion & Cabin Fire Dynamics Suite (Enhanced Edition)

An advanced interactive research dashboard and physics simulation suite built with open telemetry from NASA's **Physical Sciences Informatics (PSI)**, **NASA Open Data Portal (data.nasa.gov)**, and **NASA Technical Reports Server (NTRS)**.

---

## 🚀 Key Upgrades & New Features

### 1. 🌌 Revolving & Twinkling Deep-Space Cosmic Background
- **420+ Multi-Spectral Stars**: Features O, B, A, G, and M spectral stellar classes (cyan, white, electric blue, solar gold, warm amber, and magenta) with realistic phase-shifted sinusoidal scintillation (twinkling).
- **Revolving Celestial Orbitals**: 3D perspective rotational projection around the cosmic axis with revolving mission stations (ISS Destiny Lab, Gateway Node 1, Cygnus Free-Flyer).
- **Deep Space Nebulae**: Soft, pulsing gas clouds with drifting celestial gradients.
- **Dynamic Meteors**: Fiery shooting stars streaking randomly across the void with glowing ion tails.
- **Interactive Constellations**: Stars near the cursor connect with fine cyan constellation links.

### 2. 🪐 Dual Cybernetic Cursor & Dynamic Hover Effects
- **Dual Cursor Architecture**: High-precision cyan targeting core coupled with a smooth-lerped orbital follower ring that expands, pulses, and dashes on clickable targets.
- **Rocket Stardust Trail**: Emits fading spark particles with fluid physics as the cursor moves across the dashboard.
- **3D Card Tilt & Specular Spotlight**: All panels and flashcards compute live cursor position (`--mouse-x`, `--mouse-y`) to cast a specular reflection and tilt up to 6° in 3D perspective.

### 3. 🎴 Translucent 3D Flashcard Deck & Study Mode
- **Glassmorphism Translucency**: `backdrop-filter: blur(16px)` with layered refractive cyan borders (`rgba(0, 245, 255, 0.16)`).
- **3D Flashcard Flip Mechanics**:
  - **Front Face**: Investigation code, facility badge, research challenge question, short summary, and key telemetry metrics.
  - **Back Face**: NASA PSI discovery, fluid transport mechanism law, and spacecraft flight safety consequences.
- **Flashcard Study Deck Mode**: Step-by-step single-card study mode with **Next**, **Previous**, **Shuffle**, category filters, and keyboard shortcut (`Spacebar` to flip).
- **Multi-Category Filter & Real-Time Search**: Filter by Droplets, Gaseous Diffusion, Solids & Polymers, Large-Scale Cygnus, or Smoke Aerosols.

### 4. 🔥 Enhanced Live Flame Canvas & Interactive Thermal Probe HUD
- **Real-Time Thermal Probe**: Hovering anywhere on the flame canvas displays a holographic HUD crosshair calculating local coordinates, local gas temperature ($K$ and $°C$), airflow velocity, and oxygen transport fraction.
- **Interactive Click-to-Ignite / Perturb**: Clicking on the canvas triggers an ignition spark cluster and expanding hydrodynamic shockwave ripple.
- **Prominent Airflow Vectors**: Flow streamlines with animated direction arrows showing ECLSS diffuser injection, thermal deflection, and exhaust duct suction.

### 5. 🔊 Web Audio API Spaceflight Sound Synthesizer
- Built-in, zero-dependency audio synthesizer creating subtle futuristic tones:
  - Micro-frequency hover ticks
  - Holographic confirmation chirps
  - Flashcard whoosh flip sounds
  - Emergency warning klaxon alarms
  - Plasma spark hiss bursts
- Toggleable anytime from the topbar (`🔊 Audio: ON / 🔇 Audio: OFF`).

### 6. 📐 Interactive Dimensionless Scaling Calculator
- In **Comparative Analysis & Theory**, interactively adjust characteristic length ($L$), ventilation velocity ($U$), and gravity ($g$) to calculate:
  - **Grashof Number ($Gr$)**: Natural buoyant convection strength.
  - **Reynolds Number ($Re$)**: Forced convective inertia.
  - **Froude Number ($Fr$)**: Inertial vs. gravitational force ratio.
  - **Mixed Convection Regime Badge**: Instant prediction of whether buoyant convection, forced convection, or pure molecular diffusion dominates.

### 7. 📊 Interactive SI Ranking Leaderboard
- Search and sortable table comparing Limiting Oxygen Index (LOI), peak zero-g spread rates, and Safety Index (SI) scores for spacecraft cabin polymers, fuels, and extinguishing agents.
- Click any table header (Rank, Name, Category, SI Score) to sort ascending or descending.

### 8. 🤖 AI Assistant with Typewriter Engine & Instant Copy
- Indexed knowledge engine covering FLEX, ACME, BASS, Saffire, SAME, and SoFIE with real-time typewriter text animation and a one-click copy button.

  function getDeflectedY(x, baseY, burnerX, burnerY, flameRadius, vAir) {
    const dx = x - burnerX;
    const dy = baseY - burnerY;
    const dist = Math.sqrt(dx * dx + dy * dy);
    const influenceRadius = flameRadius * 2.35;
    if (dist < influenceRadius && vAir > 0) {
      const push = (1 - dist / influenceRadius) * (flameRadius * 0.68);
      return baseY + (dy >= 0 ? push : -push);
    }
    return baseY;
  }

  function drawProminentAirflow(p, burnerX, burnerY, flameRadius) {
    if (!showStreamlines.checked) return;

    const speedPx = p.vAir * 0.14;
    const alpha = p.vAir === 0 ? 0.18 : Math.min(0.95, 0.35 + (p.vAir / 60) * 0.6);

    for (let lane = 0; lane < NUM_STREAM_LANES; lane++) {
      const startY = 58 + (lane / (NUM_STREAM_LANES - 1)) * (canvas.height - 116);

      ctx.beginPath();
      ctx.strokeStyle =
        p.vAir === 0
          ? "rgba(0, 229, 255, 0.14)"
          : `rgba(0, 245, 255, ${alpha * 0.42})`;
      ctx.lineWidth = p.vAir > 30 ? 2.4 : 1.8;

      for (let x = 28; x <= canvas.width - 26; x += 14) {
        const rawY = startY + (x - 28) * Math.tan(p.angleRad) * 0.35;
        const clampedRawY = Math.max(46, Math.min(canvas.height - 46, rawY));
        const y = getDeflectedY(x, clampedRawY, burnerX, burnerY, flameRadius, p.vAir);
        if (x === 28) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      }
      ctx.stroke();
    }

    if (p.vAir > 0) {
      airflowParticles.forEach((pt) => {
        pt.progress += speedPx * pt.speedOffset;
        if (pt.progress > canvas.width - 34) {
          pt.progress = 30;
        }
        const x = pt.progress;
        const startY = 58 + (pt.lane / (NUM_STREAM_LANES - 1)) * (canvas.height - 116);
        const rawY = Math.max(46, Math.min(canvas.height - 46, startY + (x - 28) * Math.tan(p.angleRad) * 0.35));
        const y = getDeflectedY(x, rawY, burnerX, burnerY, flameRadius, p.vAir);

        const nextX = x + 8;
        const nextRawY = Math.max(46, Math.min(canvas.height - 46, startY + (nextX - 28) * Math.tan(p.angleRad) * 0.35));
        const nextY = getDeflectedY(nextX, nextRawY, burnerX, burnerY, flameRadius, p.vAir);
        const theta = Math.atan2(nextY - y, nextX - x);

        const arrowLen = 14 + (p.vAir / 60) * 14;

        ctx.save();
        ctx.translate(x, y);
        ctx.rotate(theta);

        ctx.strokeStyle = "#00f5ff";
        ctx.lineWidth = 2.4;
        ctx.shadowColor = "#00f5ff";
        ctx.shadowBlur = 8;
        ctx.beginPath();

        ctx.lineTo(arrowLen * 0.5, 0);
        ctx.stroke();

        ctx.fillStyle = "#e0faff";
        ctx.beginPath();
        ctx.moveTo(arrowLen * 0.5 + 5, 0);
        ctx.lineTo(arrowLen * 0.5 - 4, -4.5);
        ctx.lineTo(arrowLen * 0.5 - 4, 4.5);
        ctx.closePath();
        ctx.fill();

        ctx.restore();
      });
    } else {
      ctx.fillStyle = "rgba(0, 245, 255, 0.85)";
      ctx.font = "bold 12px sans-serif";
      ctx.fillText("VENTILATION INACTIVE (0 cm/s) — PURE MOLECULAR DIFFUSION REGIME", 230, 68);
    }
  }

  function drawFlameAndPlume(p, burnerX, burnerY, flameRadius, time) {
    const pulse = Math.sin(time * 0.006) * 2.5;
    const r = flameRadius + pulse;

    const wakeStretch = p.vAir * 1.45;
    const wakeShiftX = Math.cos(p.angleRad) * wakeStretch * 0.55;
    const wakeShiftY = Math.sin(p.angleRad) * wakeStretch * 0.35 - p.grav * 85;

    // Soot plume
    sootParticles.forEach((sp) => {
      sp.life += 0.012 + (p.vAir / 60) * 0.018;
      if (sp.life >= 1) {
        sp.life = 0;
        sp.x = burnerX + (Math.random() - 0.5) * r * 0.6;
        sp.y = burnerY + (Math.random() - 0.5) * r * 0.6;
      }
      const advX = (p.vAir * 0.085) * Math.cos(p.angleRad) + (Math.random() - 0.5) * 0.6;
      const advY = (p.vAir * 0.05) * Math.sin(p.angleRad) - p.grav * 2.8 + (Math.random() - 0.5) * 0.6;
      sp.x += advX;
      sp.y += advY;

      const alpha = (1 - sp.life) * (p.grav > 0.5 ? 0.25 : 0.52);
      ctx.fillStyle = `rgba(148, 163, 184, ${alpha})`;
      ctx.beginPath();
      ctx.arc(sp.x, sp.y, sp.size * (0.6 + sp.life), 0, Math.PI * 2);
      ctx.fill();
    });

    // Thermal Halo Field
    if (showTempField.checked) {
      const haloGrad = ctx.createRadialGradient(
        burnerX + wakeShiftX * 0.4,
        burnerY + wakeShiftY * 0.4,
        r * 0.3,
        burnerX + wakeShiftX * 0.6,
        burnerY + wakeShiftY * 0.6,
        r * 2.4
      );
      haloGrad.addColorStop(0, "rgba(255, 140, 0, 0.32)");
      haloGrad.addColorStop(0.55, "rgba(239, 68, 68, 0.12)");
      haloGrad.addColorStop(1, "rgba(239, 68, 68, 0)");
      ctx.fillStyle = haloGrad;
      ctx.beginPath();
      ctx.arc(burnerX + wakeShiftX * 0.5, burnerY + wakeShiftY * 0.5, r * 2.4, 0, Math.PI * 2);
      ctx.fill();
    }

    ctx.save();
    ctx.translate(burnerX, burnerY);

    // Luminous Flame Body Gradient
    const flameGrad = ctx.createRadialGradient(
      wakeShiftX * 0.25,
      wakeShiftY * 0.25,
      4,
      wakeShiftX * 0.5,
      wakeShiftY * 0.5,
      r * 1.45
    );

    if (p.o2 >= 28) {
      flameGrad.addColorStop(0, "rgba(255, 255, 255, 0.98)");
      flameGrad.addColorStop(0.35, "rgba(253, 224, 71, 0.94)");
      flameGrad.addColorStop(0.7, "rgba(249, 115, 22, 0.72)");
      flameGrad.addColorStop(1, "rgba(56, 189, 248, 0)");
    } else if (p.vAir === 0 && p.grav === 0) {
      // Classic 0g cool/blue spherical flamelet
      flameGrad.addColorStop(0, "rgba(224, 242, 254, 0.95)");
      flameGrad.addColorStop(0.45, "rgba(56, 189, 248, 0.78)");
      flameGrad.addColorStop(0.78, "rgba(251, 146, 60, 0.45)");
      flameGrad.addColorStop(1, "rgba(56, 189, 248, 0)");
    } else {
      flameGrad.addColorStop(0, "rgba(254, 249, 195, 0.96)");
      flameGrad.addColorStop(0.4, "rgba(251, 146, 60, 0.88)");
      flameGrad.addColorStop(0.75, "rgba(234, 88, 12, 0.58)");
      flameGrad.addColorStop(1, "rgba(59, 130, 246, 0)");
    }

    ctx.fillStyle = flameGrad;
    ctx.beginPath();
    const steps = 54;
    for (let i = 0; i <= steps; i++) {
      const a = (i / steps) * Math.PI * 2;
      const cosA = Math.cos(a);
      const sinA = Math.sin(a);

      const flowAlign = Math.max(0, Math.cos(a - p.angleRad));
      const buoyAlign = Math.max(0, -sinA);

      const localR =
        r +
        Math.pow(flowAlign, 2) * wakeStretch +
        Math.pow(buoyAlign, 2) * (p.grav * 95);

      const px = cosA * localR;
      const py = sinA * localR;
      if (i === 0) ctx.moveTo(px, py);
      else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fill();

    // Reaction Zone Boundary Ring
    ctx.strokeStyle = "rgba(56, 189, 248, 0.85)";
    ctx.lineWidth = 2.2;
    ctx.beginPath();
    ctx.arc(wakeShiftX * 0.12, wakeShiftY * 0.12, r * 0.56, 0, Math.PI * 2);
    ctx.stroke();

    // Fuel Burner / Sample Holder
    ctx.fillStyle = "#334155";
    ctx.strokeStyle = "#00f5ff";
    ctx.lineWidth = 1.6;
    ctx.beginPath();
    ctx.arc(0, 0, 11, 0, Math.PI * 2);
    ctx.fill();
    ctx.stroke();

    // Burner mount rod
    ctx.fillStyle = "#1e293b";
    ctx.fillRect(-3, 11, 6, 95);

    ctx.restore();
  }

  function drawO2Molecules(p, burnerX, burnerY, flameRadius) {
    o2Particles.forEach((m) => {
      const speed = 0.5 + p.vAir * 0.08;
      m.x += Math.cos(p.angleRad) * speed;
      m.y += Math.sin(p.angleRad) * speed * 0.45 + Math.sin(m.x * 0.03) * 0.35;

      const dx = burnerX - m.x;
      const dy = burnerY - m.y;
      const dist = Math.sqrt(dx * dx + dy * dy);
      if (dist < flameRadius * 2.2 && dist > flameRadius * 0.6) {
        m.x += (dx / dist) * 0.55;
        m.y += (dy / dist) * 0.55;
      }

      if (m.x > canvas.width - 28 || dist <= flameRadius * 0.6) {
        m.x = 32 + Math.random() * 20;
        m.y = 48 + Math.random() * (canvas.height - 96);
      }

      ctx.fillStyle = "rgba(125, 211, 252, 0.78)";
      ctx.beginPath();
      ctx.arc(m.x, m.y, m.r, 0, Math.PI * 2);
      ctx.fill();
    });
  }

  // Draw Interactive Shockwaves & Ignition Sparks
  function drawInteractiveSparks() {
    // Shockwaves
    for (let i = shockwaves.length - 1; i >= 0; i--) {
      const sw = shockwaves[i];
      sw.r += 3.2;
      sw.alpha -= 0.035;

      if (sw.alpha <= 0 || sw.r >= sw.maxR) {
        shockwaves.splice(i, 1);
        continue;
      }

      ctx.save();
      ctx.strokeStyle = `rgba(0, 245, 255, ${sw.alpha})`;
      ctx.lineWidth = 2.5;
      ctx.beginPath();
      ctx.arc(sw.x, sw.y, sw.r, 0, Math.PI * 2);
      ctx.stroke();
      ctx.restore();
    }

    // Sparks
    for (let i = ignitionSparks.length - 1; i >= 0; i--) {
      const sp = ignitionSparks[i];
      sp.x += sp.vx;
      sp.y += sp.vy;
      sp.vx *= 0.94;
      sp.vy *= 0.94;
      sp.life -= sp.decay;

      if (sp.life <= 0) {
        ignitionSparks.splice(i, 1);
        continue;
      }

      ctx.save();
      ctx.fillStyle = sp.color;
      ctx.shadowColor = sp.color;
      ctx.shadowBlur = 8;
      ctx.globalAlpha = sp.life;
      ctx.beginPath();
      ctx.arc(sp.x, sp.y, sp.size * sp.life, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
    }
  }

  // Draw Interactive Temperature Probe HUD
  function drawTemperatureProbeHUD(p, burnerX, burnerY, flameRadius, peakTemp) {
    if (!canvasMouse.isOver) return;

    const mx = canvasMouse.x;
    const my = canvasMouse.y;

    const dx = mx - burnerX;
    const dy = my - burnerY;
    const dist = Math.sqrt(dx * dx + dy * dy);

    // Compute local physical temperature at probe
    let localTemp = 295; // ambient K
    const ambientK = 295;
    if (dist < flameRadius * 0.95) {
      localTemp = peakTemp;
    } else {
      // Exponential decay outwards
      const decayFactor = Math.exp(-(dist - flameRadius * 0.95) / (flameRadius * 1.2));
      // Thermal wake enhancement in flow direction
      const flowDot = (dx * Math.cos(p.angleRad) + dy * Math.sin(p.angleRad)) / (dist || 1);
      const wakeBoost = Math.max(0, flowDot) * (p.vAir / 60) * 0.45;
      localTemp = Math.round(ambientK + (peakTemp - ambientK) * decayFactor * (1 + wakeBoost));
    }

    const localCelsius = (localTemp - 273.15).toFixed(0);

    // 1. Crosshair Target Reticle
    ctx.save();
    ctx.strokeStyle = "#00f5ff";
    ctx.lineWidth = 1.4;
    ctx.shadowColor = "#00f5ff";
    ctx.shadowBlur = 6;

    // Outer crosshair circle
    ctx.beginPath();
    ctx.arc(mx, my, 14, 0, Math.PI * 2);
    ctx.stroke();

    // Center dot
    ctx.fillStyle = "#ffffff";
    ctx.beginPath();
    ctx.arc(mx, my, 2, 0, Math.PI * 2);
    ctx.fill();

    // Corner brackets
    const arm = 18;
    ctx.beginPath();
    ctx.moveTo(mx - arm, my);
    ctx.lineTo(mx - 6, my);
    ctx.moveTo(mx + 6, my);
    ctx.lineTo(mx + arm, my);
    ctx.moveTo(mx, my - arm);
    ctx.lineTo(mx, my - 6);
    ctx.moveTo(mx, my + 6);
    ctx.lineTo(mx, my + arm);
    ctx.stroke();

    // 2. Translucent Telemetry Box
    const boxW = 155;
    const boxH = 68;
    let boxX = mx + 24;
    let boxY = my - 34;

    if (boxX + boxW > canvas.width - 10) boxX = mx - boxW - 24;
    if (boxY + boxH > canvas.height - 10) boxY = canvas.height - boxH - 12;
    if (boxY < 10) boxY = 12;

    ctx.fillStyle = "rgba(10, 17, 34, 0.88)";
    ctx.strokeStyle = "rgba(0, 245, 255, 0.6)";
    ctx.lineWidth = 1.2;
    ctx.beginPath();
    if (ctx.roundRect) {
      ctx.roundRect(boxX, boxY, boxW, boxH, 6);
    } else {
      ctx.rect(boxX, boxY, boxW, boxH);
    }
    ctx.fill();
    ctx.stroke();

    // Telemetry text
    ctx.fillStyle = "#00f5ff";
    ctx.font = "bold 10px monospace";
    ctx.fillText("THERMAL PROBE // HUD", boxX + 10, boxY + 16);

    ctx.fillStyle = localTemp > 800 ? "#ff9e00" : "#e2e8f0";
    ctx.font = "bold 13px sans-serif";
    ctx.fillText(`${localTemp} K (${localCelsius}°C)`, boxX + 10, boxY + 34);

    ctx.fillStyle = "#94a3b8";

    let n = ix * 374761393 + iy * 668265263;

  function valueNoise2D(x, y, periodX) {
    const ix = Math.floor(x);
    const iy = Math.floor(y);
    const fx = x - ix;
    const fy = y - iy;
    const ux = fx * fx * (3 - 2 * fx);
    const uy = fy * fy * (3 - 2 * fy);

    const ix0 = periodX ? ((ix % periodX) + periodX) % periodX : ix;
    const ix1 = periodX ? (((ix + 1) % periodX) + periodX) % periodX : ix + 1;

    const a = hash2D(ix0, iy);
    const b = hash2D(ix1, iy);
    const c = hash2D(ix0, iy + 1);
    const d = hash2D(ix1, iy + 1);

    return a + (b - a) * ux + (c - a) * uy + (a - b - c + d) * ux * uy;

      s.y = rawY - parallaxY * (s.z * 1.2);
